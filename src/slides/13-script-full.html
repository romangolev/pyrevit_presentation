# RenameSheets.pushbutton/script.py
#! python3

from pyrevit import revit, DB, forms, script

logger = script.get_logger()
doc    = revit.doc

# ── 1. Ask user for a find / replace pair ──────────────────────
result = forms.ask_for_string(
    prompt="Find text in sheet names:",
    title="Rename Sheets"
)
if not result:
    script.exit()

find, replace = result.split("→") if "→" in result else (result, "")

# ── 2. Collect all sheets ──────────────────────────────────────
sheets = DB.FilteredElementCollector(doc) \
           .OfClass(DB.ViewSheet) \
           .ToElements()

# ── 3. Rename inside a transaction ────────────────────────────
with revit.Transaction("Rename Sheets"):
    count = 0
    for sheet in sheets:
        name = sheet.Name
        if find in name:
            sheet.Name = name.replace(find, replace)
            count += 1

logger.info("Renamed {} sheets.".format(count))
forms.alert("Done — renamed {} sheets.".format(count), title="Rename Sheets")
#! python3 tells PyRevit to use the CPython engine. revit.Transaction is a context manager — commits on exit, rolls back on exception.
